from database.DB import ManagerBase
import logging

class Absences_model(ManagerBase):
    def __init__(self):
        super().__init__()
        
    def creer_table_absences(self):
        self.cusor.execute("""
            CREATE TABLE IF NOT EXISTS absences (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                student_id INTEGER NOT NULL,
                date TEXT NOT NULL,
                status TEXT NOT NULL CHECK(status IN ('Justifiée', 'Non justifiée')),
                FOREIGN KEY (student_id) REFERENCES students(id)
            );
        """)
        self.connecte.commit()

    def ajouter_absence(self, student_id, date, status="Non justifiée"):
        statuts_autorises = ["Justifiée", "Non justifiée"]
        if status not in statuts_autorises:
            print(" Erreur : Le statut doit être 'Justifiée' ou 'Non justifiée'.")
            return False 
            
        self.cusor.execute("""
            INSERT INTO absences (student_id, date, status) 
            VALUES (?, ?, ?);
        """, (student_id, date, status))
        self.connecte.commit()
        logging.info(f"Absence enregistrée pour l'élève ID {student_id} le {date} ({status})")
        print(f" Absence enregistrée le {date} pour l'étudiant ID {student_id} ({status}) !")
        return True

    def supprimer_absence(self, id_absence):
        self.cusor.execute("DELETE FROM absences WHERE id = ?", (id_absence,))
        self.cusor.execute("DELETE FROM sqlite_sequence WHERE name='absences'")
        self.connecte.commit()
        logging.info(f"Absence supprimée, ID : {id_absence}")
        print(" Absence supprimée avec succès !")

    def modifier_absence_status(self, id_absence, status):
        statuts_autorises = ["Justifiée", "Non justifiée"]
        if status not in statuts_autorises:
            print(" Erreur : Statut invalide.")
            return False

        self.cusor.execute("""
            UPDATE absences 
            SET status = ?
            WHERE id = ?
        """, (status, id_absence))
        self.connecte.commit()
        logging.info(f"Statut de l'absence ID {id_absence} basculé sur : {status}")
        print(" Le statut de l'absence a été mis à jour !")
        return True

    def consulter_historique_etudiant(self, student_id):
        self.cusor.execute("""
            SELECT * FROM absences WHERE student_id = ? ORDER BY date DESC
        """, (student_id,))
        logging.info(f"Consultation de l'historique d'absences de l'élève ID {student_id}")
        return self.cusor.fetchall()

    def supprimer_tous_les_absences(self):
        self.cusor.execute("DELETE FROM absences")
        self.cusor.execute("DELETE FROM sqlite_sequence WHERE name='absences'")
        self.connecte.commit()
        logging.info("Toutes les absences ont été supprimées.")
        print(" Toutes les absences ont été supprimées.")

    def rechercher_absence(self, id_absence):
        self.cusor.execute("SELECT * FROM absences WHERE id = ?", (id_absence,))
        return self.cusor.fetchone()

    def liste_tout_absence(self):
        self.cusor.execute("SELECT * FROM absences")
        return self.cusor.fetchall()

    def close(self):
        self.connecte.close()



from database.DB import ManagerBase
import logging

class Notes_model(ManagerBase):
    def __init__(self):
        super().__init__()
        
    def creer_table_Notes(self):
        self.cusor.execute("""
            CREATE TABLE IF NOT EXISTS grades (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                student_id INTEGER NOT NULL,
                subject_id INTEGER NOT NULL,
                note REAL NOT NULL CHECK(note >= 0 AND note <= 20),
                FOREIGN KEY (student_id) REFERENCES students(id),
                FOREIGN KEY (subject_id) REFERENCES subjects(id)
            );
        """)
        self.connecte.commit()

    def ajouter_note(self, student_id, subject_id, note):
        if not (0 <= note <= 20):
            logging.warning(f"Tentative d'ajout d'une note hors limites (0-20) : {note}")
            print("Erreur : La note doit être comprise entre 0 et 20.")
            return False 
            
        self.cusor.execute("""
            INSERT INTO grades (student_id, subject_id, note) 
            VALUES (?, ?, ?);
        """, (student_id, subject_id, note))
        self.connecte.commit()
        logging.info(f"Note de {note}/20 ajoutée pour l'étudiant ID {student_id} (Matière ID {subject_id})")
        print(f"Note de {note}/20 ajoutée avec succès pour l'étudiant ID {student_id} !")
        return True

    def supprimer_note(self, id_note):
        self.cusor.execute("DELETE FROM grades WHERE id = ?", (id_note,))
        self.cusor.execute("DELETE FROM sqlite_sequence WHERE name='grades'")
        self.connecte.commit()
        logging.info(f"Note supprimée, ID : {id_note}")
        print(" Note supprimée avec succès !")

    def modifier_note_valeur(self, id_note, nouvelle_note):
        if not (0 <= nouvelle_note <= 20):
            logging.warning(f"Tentative de modification avec une note hors limites : {nouvelle_note}")
            print("Erreur : La note doit être comprise entre 0 et 20.")
            return False 

        self.cusor.execute("""
            UPDATE grades 
            SET note = ?
            WHERE id = ?
        """, (nouvelle_note, id_note))
        self.connecte.commit()
        logging.info(f"Note ID {id_note} modifiée. Nouvelle valeur : {nouvelle_note}/20")
        print(" Note mise à jour avec succès !")
        return True

    def calculer_moyenne_etudiant(self, student_id):
        self.cusor.execute("""
            SELECT AVG(note) FROM grades WHERE student_id = ?
        """, (student_id,))
        resultat = self.cusor.fetchone()
        moyenne = resultat[0] if resultat[0] is not None else 0.0
        logging.info(f"Calcul de moyenne pour l'étudiant ID {student_id} : {moyenne}")
        return moyenne

    def supprimer_tous_les_notes(self):
        self.cusor.execute("DELETE FROM grades")
        self.cusor.execute("DELETE FROM sqlite_sequence WHERE name='grades'")
        self.connecte.commit()
        logging.info("Toutes les notes ont été purgées.")
        print(" Toutes les notes ont été supprimées.")

    def rechercher_note(self, id_note):
        self.cusor.execute("SELECT * FROM grades WHERE id = ?", (id_note,))
        return self.cusor.fetchone()

    def liste_tout_note(self):
        self.cusor.execute("SELECT * FROM grades")
        return self.cusor.fetchall()

    def close(self):
        self.connecte.close()





from database.DB import ManagerBase
import logging

class Sujet_matiere(ManagerBase):
    def __init__(self):
        super().__init__()
    
    def creer_table_matiere(self):
        self.cusor.execute("""
            CREATE TABLE IF NOT EXISTS subjects (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                nom TEXT NOT NULL,                                  
                teacher_id INTEGER,
                FOREIGN KEY (teacher_id) REFERENCES teachers(id)
                ON DELETE SET NULL
            );
        """)
        self.connecte.commit()
        
    def ajouter_matiere(self, nom, teacher_id=None):
        if not nom or nom.strip() == "":
            print(" Erreur : Le nom de la matière ne peut pas être vide.")
            return False
            
        self.cusor.execute("""
            INSERT INTO subjects (nom, teacher_id) 
            VALUES (?, ?);
        """, (nom, teacher_id))
        self.connecte.commit()
        logging.info(f"Matière ajoutée : {nom}")
        print(f" Matière '{nom}' ajoutée avec succès !")
        return True
        
    def supprimer_matiere(self, id_matiere):
        self.cusor.execute("DELETE FROM subjects WHERE id = ?", (id_matiere,))
        self.cusor.execute("DELETE FROM sqlite_sequence WHERE name='subjects'")
        self.connecte.commit()
        logging.info(f"Matière supprimée, ID : {id_matiere}")
        print(" Matière supprimée avec succès !")
    
    def modifier_matiere_nom(self, id_matiere, nom):
        self.cusor.execute("""
            UPDATE subjects

        print(" Nom de la matière mis à jour !")

    def affecter_professeur(self, id_matiere, teacher_id):

        self.connecte.commit()
        logging.info(f"Professeur ID {teacher_id} affecté à la matière ID {id_matiere}")
        print(" Professeur affecté à la matière avec succès !")

    def supprimer_tous_les_matieres(self):

        self.cusor.execute("DELETE FROM sqlite_sequence WHERE name='subjects'")
        self.connecte.commit()
        logging.info("Toutes les matières ont été supprimées.")
        print(" Toutes les matières ont été supprimées.")
        
    def rechercher_matiere(self, id_matiere):
        self.cusor.execute("SELECT * FROM subjects WHERE id = ?", (id_matiere,))
        return self.cusor.fetchone()
    
    def liste_tout_matiere(self):
        self.cusor.execute("SELECT * FROM subjects")
        return self.cusor.fetchall()

        self.connecte.close()
